# Python 3.12
apt update
apt install -y software-properties-common
add-apt-repository -y ppa:deadsnakes/ppa
apt install -y python3.12 python3.12-venv

# окружение (одно на все треки)
python3.12 -m venv .venv
source .venv/bin/activate
pip install -U pip
pip install -r requirements.txt

# ядро для Jupyter (в ноутбуке: Kernel -> Change kernel -> Brain-Decoding (3.12))
python -m ipykernel install --user --name brain-decoding --display-name "Brain-Decoding (3.12)"

# данные (скачивание с NEMAR повторяется при таймаутах и продолжается с места обрыва)
cd track2_bci && python -m bci.data --data-dir data && cd ..
# Track 1: Alljoined-1.6M под лицензией CC BY-NC-ND, переменная ниже означает ваше согласие с ней; нужен GPU для DINOv2
export ALLJOINED_ACCEPT_LICENCE=1
cd track1_image && python -m image_decoding.data --study xu2025alljoined --data-dir data && cd ..
# Track 3: ~1.2 ГБ
cd track3_sleep && python -m sleep_onset.data --data-dir data && cd ..
# Track 4: ~330 ГБ загрузки и ~440 ГБ кэша
cd track4_emg && python -m emg_pose.data --data-dir data && cd ..

# отправки: ZIP в <трек>/submissions/, загружать в Codabench -> My Submissions
cd track2_bci && python -m bci.make_submission && cd ..
cd track1_image && python -m image_decoding.make_submission --study xu2025alljoined && cd ..
cd track3_sleep && python -m sleep_onset.make_submission && cd ..
cd track4_emg && python -m emg_pose.make_submission --seeds 3 --epochs 40 && cd ..
